?item.right.state = 3
    equipR fissure
    equip @item.right@


func CanUse(weap)
    return item.GetCooldown(weap) <= 0 & item.CanActivate()

func Use(weap)
    CA.UseR(weap)

func InRange(num, min, max)
    return num >= min & num <= max
    
func Dash()
    var FD = foe.distance
    
    ?FD > 6 & CanUse("quarterstaff")
        equip quarterstaff
        activate R
        return false
    :?InRange(FD, 11, 16)
        ?CanUse("bash")
            equip bashing shield
        :?CanUse("dash")
            equip dashing shield
        :
            return true
        return false
    :
        return true

func Dash2()
    var FD = foe.distance
    
    ?InRange(FD, 11, 16)
        ?CanUse("bash")
            Use("bash")
        :?CanUse("dash")
            Use("dash")
        :
            return true
        return false
    :
        return true

func MoonJuggle(Weap1, Weap2, moon_del)
    ?totaltime % moon_del = 0
        equipL @Weap2@
        equipR @Weap1@
    ?totaltime % moon_del = 1
        equipL moondial
        equipR @Weap1@
    ?totaltime % moon_del = 2
        equipR @Weap1@
        equipL @Weap2@
    ?totaltime % moon_del = 3
        equipL moondial
        equipR @Weap2@
