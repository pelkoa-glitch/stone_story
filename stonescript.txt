func CanUse(weap)
    return item.GetCooldown(weap) <= 0 & item.CanActivate()

func Use(weap)
    CA.UseR(weap)

func InRange(num, min, max)
    return num >= min & num <= max
    
func Dash()
    var FD = foe.distance
    
    ?FD > 6 & CanUse(＂quarterstaff＂)
        equip quarterstaff
        activate R
        return false
    :?InRange(FD, 11, 16)
        ?CanUse(＂bash＂)
            equip bashing shield
        :?CanUse(＂dash＂)
            equip dashing shield
        :
            return true
        return false
    :
        return true

func Dash2()
    var FD = foe.distance
    
    ?InRange(FD, 11, 16)
        ?CanUse(＂bash＂)
            Use(＂bash＂)
        :?CanUse(＂dash＂)
            Use(＂dash＂)
        :
            return true
        return false
    :
        return true

func MoonJuggle(Weap1, Weap2)
    ?totaltime % 4 = 0
        equipL @Weap2@
        equipR @Weap1@
    ?totaltime % 4 = 1
        equipL moondial
        equipR @Weap1@
    ?totaltime % 4 = 2
        equipR @Weap1@
        equipL @Weap2@
    ?totaltime % 4 = 3
        equipL moondial
        equipR @Weap2@

var blade_frame = 0

func DoBladeScroll()
    var cdb = item.GetCooldown(＂blade＂)
    ? cdb <= 0 & blade_frame <= 0
        blade_frame = 1
        screen.Next()
        return true 
    

    : ? blade_frame > 0
        blade_frame++
        ? blade_frame >= 30
            blade_frame = 0
            equip blade
            activate R
            screen.ResetOffset()
            return true 
        return true 
        
    return false
